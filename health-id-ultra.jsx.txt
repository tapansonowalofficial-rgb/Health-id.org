/* ── Neural Particle Canvas ── */
function NeuralCanvas() {
  const ref = useRef();
  useEffect(() => {
    const c = ref.current; if (!c) return;
    const ctx = c.getContext("2d");
    let W, H, nodes = [], frame;
    const resize = () => {
      W = c.width = c.offsetWidth;
      H = c.height = c.offsetHeight;
    };
    resize();
    window.addEventListener("resize", resize);
    for (let i = 0; i < 55; i++) nodes.push({
      x: Math.random() * W, y: Math.random() * H,
      vx: (Math.random() - 0.5) * 0.3, vy: (Math.random() - 0.5) * 0.3,
      r: Math.random() * 1.5 + 0.5,
      pulse: Math.random() * Math.PI * 2,
      type: Math.random() > 0.7 ? "plasma" : "cyan",
    });
    const draw = () => {
      ctx.clearRect(0, 0, W, H);
      nodes.forEach(n => {
        n.x += n.vx; n.y += n.vy; n.pulse += 0.02;
        if (n.x < 0 || n.x > W) n.vx *= -1;
        if (n.y < 0 || n.y > H) n.vy *= -1;
      });
      nodes.forEach((a, i) => nodes.slice(i + 1).forEach(b => {
        const d = Math.hypot(a.x - b.x, a.y - b.y);
        if (d < 130) {
          const alpha = (1 - d / 130) * 0.35;
          const g = ctx.createLinearGradient(a.x, a.y, b.x, b.y);
          g.addColorStop(0, `rgba(0,245,255,${alpha})`);
          g.addColorStop(1, `rgba(191,95,255,${alpha * 0.5})`);
          ctx.strokeStyle = g; ctx.lineWidth = 0.6;
          ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke();
        }
      }));
      nodes.forEach(n => {
        const glow = Math.sin(n.pulse) * 0.5 + 0.5;
        const color = n.type === "plasma" ? "191,95,255" : "0,245,255";
        const gr = ctx.createRadialGradient(n.x, n.y, 0, n.x, n.y, n.r * 4);
        gr.addColorStop(0, `rgba(${color},${0.9 * glow})`);
        gr.addColorStop(1, `rgba(${color},0)`);
        ctx.fillStyle = gr;
        ctx.beginPath(); ctx.arc(n.x, n.y, n.r * 4, 0, Math.PI * 2); ctx.fill();
        ctx.fillStyle = `rgba(${color},1)`;
        ctx.beginPath(); ctx.arc(n.x, n.y, n.r, 0, Math.PI * 2); ctx.fill();
      });
      frame = requestAnimationFrame(draw);
    };
    draw();
    return () => { cancelAnimationFrame(frame); window.removeEventListener("resize", resize); };
  }, []);
  return <canvas ref={ref} style={{ position: "fixed", inset: 0, width: "100%", height: "100%", zIndex: 0, opacity: 0.6 }} />;
}

/* ── QR Generator ── */
function RealQR({ data, size = 150 }) {
  const cells = [];
  let hash = 0;
  for (let i = 0; i < data.length; i++) hash = ((hash << 5) - hash) + data.charCodeAt(i);
  const grid = 21;
  const cell = size / grid;
  const pat = Array.from({ length: grid }, (_, r) =>
    Array.from({ length: grid }, (_, c) => {
      if ((r < 7 && c < 7) || (r < 7 && c > grid - 8) || (r > grid - 8 && c < 7)) return 1;
      if ((r === 7 && c < 8) || (r < 8 && c === 7) || (r === 7 && c > grid - 9) || (r < 8 && c === grid - 8) || (r > grid - 9 && c === 7) || (r === grid - 8 && c < 8)) return 1;
      if (r === 6 && c > 7 && c < grid - 8) return c % 2 === 0 ? 1 : 0;
      if (c === 6 && r > 7 && r < grid - 8) return r % 2 === 0 ? 1 : 0;
      const seed = (r * grid + c + hash) % 7;
      return seed < 3 ? 1 : 0;
    })
  );
  pat.forEach((row, r) => row.forEach((v, c) => {
    if (v) cells.push(<rect key={`${r}-${c}`} x={c * cell} y={r * cell} width={cell - 0.5} height={cell - 0.5} fill={C.cyan} rx={0.5} />);
  }));
  return (
    <svg width={size} height={size} style={{ borderRadius: 8 }}>
      <rect width={size} height={size} fill={C.deep} rx={8} />
      {cells}
      <rect x={size * 0.38} y={size * 0.38} width={size * 0.24} height={size * 0.24} rx={4} fill={C.deep} />
      <text x={size * 0.5} y={size * 0.54} textAnchor="middle" fontSize={size * 0.1} fill={C.cyan} fontFamily="monospace">🧠</text>
    </svg>
  );
}

/* ── Vitals Chart ── */
function VitalChart({ data, color, label }) {
  const max = Math.max(...data), min = Math.min(...data);
  const H = 48, W = 120;
  const pts = data.map((v, i) => `${(i / (data.length - 1)) * W},${H - ((v - min) / (max - min + 1)) * H * 0.85 - 4}`).join(" ");
  return (
    <div>
      <svg width={W} height={H}>
        <defs>
          <linearGradient id={`g${label}`} x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor={color} stopOpacity="0.3" />
            <stop offset="100%" stopColor={color} stopOpacity="0" />
          </linearGradient>
        </defs>
        <polyline points={`0,${H} ${pts} ${W},${H}`} fill={`url(#g${label})`} stroke="none" />
        <polyline points={pts} fill="none" stroke={color} strokeWidth="1.5" strokeLinejoin="round" />
        {data.map((v, i) => i === data.length - 1 && (
          <circle key={i} cx={(i / (data.length - 1)) * W} cy={H - ((v - min) / (max - min + 1)) * H * 0.85 - 4}
            r={3} fill={color} style={{ filter: `drop-shadow(0 0 4px ${color})` }} />
        ))}
      </svg>
    </div>
  );
}

function BootScreen({ onDone }) {
  const [step, setStep] = useState(0);
  const steps = ["INITIALIZING NEURAL CORE...", "LOADING HEALTH DATABASE...", "ENCRYPTING PATIENT DATA...", "CALIBRATING AI MODELS...", "SYSTEM ONLINE"];
  useEffect(() => {
    const t = setInterval(() => setStep(s => { if (s >= steps.length - 1) { clearInterval(t); setTimeout(onDone, 600); return s; } return s + 1; }), 500);
    return () => clearInterval(t);
  }, []);
  return (
    <div style={{ position: "fixed", inset: 0, background: C.void, display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center", zIndex: 9999, gap: 24 }}>
      <NeuralBrain size={120} />
      <div style={{ fontFamily: "'Orbitron', monospace", fontSize: 24, fontWeight: 900, background: `linear-gradient(135deg, ${C.cyan}, ${C.neon})`, WebkitBackgroundClip: "text", WebkitTextFillColor: "transparent", letterSpacing: "0.15em" }}>NEURAL HEALTH ID</div>
      <div style={{ width: 280 }}>
        {steps.slice(0, step + 1).map((s, i) => (
          <div key={i} style={{ fontFamily: "'Share Tech Mono', monospace", fontSize: 11, color: i === step ? C.neon : C.textDim, marginBottom: 6, display: "flex", alignItems: "center", gap: 8 }}>
            <span style={{ color: i === step ? C.neon : C.textDim }}>{i < step ? "✓" : i === step ? "►" : "○"}</span> {s}
          </div>
        ))}
      </div>
      <div style={{ width: 280, height: 3, background: `${C.border}`, borderRadius: 2, overflow: "hidden" }}>
        <div style={{ width: `${((step + 1) / steps.length) * 100}%`, height: "100%", background: `linear-gradient(90deg, ${C.cyan}, ${C.neon})`, borderRadius: 2, transition: "width 0.4s ease", boxShadow: `0 0 10px ${C.cyanGlow}` }} />
      </div>
    </div>
  );
}
